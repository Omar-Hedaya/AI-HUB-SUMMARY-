Precedence Rules
* / % precede + -

Calculated Values

EXECUTION PLAN & SOLUTION

Multiplication 2*3=6 runs first, so 8+6=14. Parentheses force subtraction before division for b, and division/multiplication associate left-to-right for d.

    int c = 31 % 3;                  // 31 mod 3 = 1
    float d = (4 + 2) / 4.0 * 8;     // 6 / 4.0 * 8 = 1.5 * 8 = 12.0

    cout << "a = " << a << endl;
    cout << "b = " << b << endl;
    cout << "c = " << c << endl;
    cout << "d = " << d << endl;
    return 0;
}

int main() {
    double num1, num2;
    char op;
    cout << "Enter two numbers and operator (+, -, *, /): ";

    switch (op) {
        case '+':
            cout << "Result = " << num1 + num2 << endl;
            break;
        case '-':
            cout << "Result = " << num1 - num2 << endl;
            break;

            else
                cout << "Error: Division by zero" << endl;
            break;
        default:
            cout << "Invalid operator" << endl;
    }
    return 0;
}

        cout << "Enter PIN (Attempt " << attempt << " of 3): ";
        cin >> enteredPin;

        if (enteredPin == CORRECT_PIN) {
            cout << "Access granted" << endl;
            accessGranted = true;
            break;
        } else {
            if (attempt < 3) {
                cout << "Incorrect PIN. Try again." << endl;
            }
        }
    }

    if (!accessGranted) {
        cout << "Account locked" << endl;
    }

    return 0;
}
